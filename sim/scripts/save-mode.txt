# Save mode and the cell probe: with the radio running the child can live on the radio alone; the
# mobile network is switched on for a probe of three minutes and goes off again by itself.
advance 5s
# Without a radio there is no way out of the device, so save mode is refused.
savemode on
see notice.save-mode-needs-radio
check savemode off
check cellular on

radio start
advance 2s
radio peer NodeOne
advance 1s
savemode on
advance 1s
check savemode on
check cellular off
see notice.save-mode-on

# A probe looks for the network for three minutes.
probe
advance 1s
check cellular on
advance 170s
check cellular on
advance 20s
check cellular off
check savemode on

# Leaving save mode brings the network back for good.
savemode off
advance 2s
check savemode off
check cellular on
